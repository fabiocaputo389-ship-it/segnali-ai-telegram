async function safeFetchJson(url) { const res = await fetch(url, { cache: "no-store" }); const rawText = await res.text(); if (!rawText || !rawText.trim()) { throw new Error(`Risposta vuota da ${url} (HTTP ${res.status})`); } let parsed; try { parsed = JSON.parse(rawText); } catch (_e) { throw new Error(`Risposta non JSON da ${url} (HTTP ${res.status}): ${rawText.slice(0, 120)}`); } if (!res.ok && !parsed.error) { parsed.error = `HTTP ${res.status} su ${url}`; } return parsed; } async function refreshAll() { try { const [sigData, accData, statusData] = await Promise.all([ safeFetchJson(`/api/signals?${getQueryParams()}`), safeFetchJson("/api/account/summary"), safeFetchJson("/api/autotrade/status"), ]); const realMode = Boolean(sigData.real_mode ?? accData.real_mode ?? true); const apiOk = Boolean(sigData.performance?.api_configured ?? accData.api_configured); const killHit = Boolean(sigData.kill_switch_hit); if (killHit && !lastKillState) { playTone(260, 0.35); } lastKillState = killHit; const modePill = document.getElementById("modePill"); modePill.textContent = realMode ? "REAL MODE" : "SIM MODE"; modePill.className = realMode ? "pill ok" : "pill warn"; const apiPill = document.getElementById("apiPill"); apiPill.textContent = apiOk ? "BINGX API OK" : "BINGX API MISSING"; apiPill.className = apiOk ? "pill ok" : "pill err"; const killPill = document.getElementById("killPill"); killPill.className = killHit ? "pill err" : "pill hidden"; updateAutoTradeButton(Boolean(sigData.auto_trade ?? accData.auto_trade)); const bal = sigData.account_balance || accData.balance; if (bal && bal.equity !== undefined) { document.getElementById("kpiEquity").textContent = `$${fmtNum(bal.equity, 2)}`; document.getElementById("kpiMargin").textContent = `Margine disp.: $${fmtNum(bal.available_margin, 2)}`; } else { const fbBal = Number(document.getElementById("balanceInput").value || 2.0); document.getElementById("kpiEquity").textContent = `$${fmtNum(fbBal, 2)}`; document.getElementById("kpiMargin").textContent = accData.balance_error || "Saldo fallback attivo"; } const daily = sigData.performance?.daily || {}; const monthly = sigData.performance?.monthly || {}; const dPnl = Number(daily.net_pnl || 0); const mPnl = Number(monthly.net_pnl || 0); const dEl = document.getElementById("kpiDailyPnl"); dEl.textContent = fmtUsd(dPnl); dEl.className = `kpi-val ${dPnl >= 0 ? Auto-Trade disabilitato.`, "err"); } else { setBanner("", "warn"); } const monitorRows = sigData.market_monitoring_table || []; const historyRows = statusData.history || []; renderPriceChart(monitorRows); renderPnlChart(historyRows); renderSignals(signals); renderPositions(positions); renderMonitor(monitorRows); renderHistory(historyRows); renderLogs(tel.telemetry_logs || []); } catch (err) { setBanner(`Errore aggiornamento terminale: ${err.message}`, "err"); } }
